import * as React from 'react';
/** Loading placeholder. variant='text' (n lines) | 'card' (bordered article card) | 'article' (headline + byline + body). Sunken-tone shimmer, no grey-on-white. */
export interface SkeletonProps{
  variant?:'text'|'card'|'article';
  /** body lines. @default 3 */
  lines?:number;
  width?:number|string;
}
export declare function Skeleton(props:SkeletonProps):React.ReactElement;
