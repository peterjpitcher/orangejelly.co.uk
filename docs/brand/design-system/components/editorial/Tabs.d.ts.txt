import * as React from 'react';
/** Underline tabs (orange active rule). Survives the rebuild for in-page content switches only — never as primary navigation. */
export interface TabsProps{
  items?:Array<{label:string;content?:React.ReactNode}>;
  /** controlled index; omit for internal state */
  active?:number;
  onChange?:(i:number)=>void;
  /** set false when panels render elsewhere */
  renderPanel?:boolean;
}
export declare function Tabs(props:TabsProps):React.ReactElement;
