import * as React from 'react';
/**
 * Ink site footer: brand + tagline, link columns, legal bar with signature line.
 */
export interface FooterProps {
  /** Brand node; pass the white reversed logo on ink. Defaults to the type wordmark. */
  logo?: React.ReactNode;
  /** @default 'You bring the growth problem. We build the solution.' */
  tagline?: React.ReactNode;
  columns?: Array<{ title: string; links: Array<{ label: string; href: string }> }>;
  /** @default '© <year> Orange Jelly Limited' */
  legal?: React.ReactNode;
  /** Right side of the legal bar. @default 'AI is part of the toolkit, not the product.' */
  note?: React.ReactNode;
  /** Extra brand-block content (e.g. availability Tag, contact link). */
  children?: React.ReactNode;
}
export declare function Footer(props: FooterProps): JSX.Element;
