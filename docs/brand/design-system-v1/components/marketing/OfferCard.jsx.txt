import React from 'react';
import {Card} from '../content/Card.jsx';
import {Button} from '../core/Button.jsx';
export function OfferCard({eyebrow,name,blurb,includes=[],footnote,cta,featured=false}){
  const dark=featured;
  return <Card tone={dark?'ink':'paper'} style={{display:'flex',flexDirection:'column',gap:14,height:'100%'}}>
    {eyebrow&&<span className="oj-eyebrow" style={{color:dark?'var(--oj-peach)':'var(--text-accent)'}}>{eyebrow}</span>}
    <span style={{fontFamily:'var(--font-display)',fontWeight:900,fontSize:25,lineHeight:1.05,letterSpacing:'-.02em'}}>{name}</span>
    {blurb&&<p style={{margin:0,fontSize:14.5,lineHeight:1.55,color:dark?'rgba(247,245,241,.78)':'var(--text-secondary)'}}>{blurb}</p>}
    {includes.length>0&&<ul style={{listStyle:'none',margin:0,padding:0,display:'flex',flexDirection:'column',gap:8,flex:1}}>
      {includes.map((x,i)=><li key={i} style={{display:'flex',gap:10,fontSize:14.5,lineHeight:1.45}}><span aria-hidden="true" style={{color:'var(--oj-orange)',fontWeight:700}}>→</span><span>{x}</span></li>)}
    </ul>}
    {cta&&<div><Button variant={dark?'primary':'ink'} arrow href={cta.href} onClick={cta.onClick}>{cta.label}</Button></div>}
    {footnote&&<span style={{fontSize:12.5,color:dark?'rgba(247,245,241,.55)':'var(--text-muted)'}}>{footnote}</span>}
  </Card>;
}
